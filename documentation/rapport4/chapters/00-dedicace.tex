\chapter*{Dédicace}
\addcontentsline{toc}{chapter}{Dédicace}
\thispagestyle{empty}
\vspace*{2cm}
\begin{center}
\textit{À l'âme de mon père Karim,\\
toi qui m'as transmis le goût de l'effort et de la persévérance,\\
et dont la mémoire continue de guider chacun de mes pas.}\\[0.8cm]

\textit{À ma mère Samia,\\
pour ton amour inconditionnel, ton écoute et ta patience infinies\\
qui m'ont accompagné jusqu'à l'aboutissement de ce parcours.}\\[0.8cm]

\textit{À ma sœur Mariem,\\
pour ta présence rassurante et ton soutien de chaque instant,\\
dans les moments d'effort comme dans ceux de doute.}\\[0.8cm]

\textit{À tous mes amis et proches,\\
pour vos encouragements constants et les souvenirs partagés\\
qui ont rendu ces années si précieuses.}\\[1cm]

\textit{Je vous dédie ce travail.}
\end{center}
